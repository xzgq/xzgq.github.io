# 2. 用户部分（tom）

## 2.1 环境设置

```bash
REMOTE=https://gitee.com/mirrors/oh-my-zsh.git
git clone $REMOTE
pushd oh-my-zsh/tools > /dev/null
REMOTE=$REMOTE sh install.sh
# 安装过程会启动zsh，exit或Ctrl+D退出
popd > /dev/null
rm -rf oh-my-zsh
# 配置oh-my-zsh的风格主题
sed -i -e "s/\(ZSH_THEME=\).*/\1\"darkblood\"/" ~/.zshrc
# 配置oh-my-zsh自动更新
sed -i -e "s/# \(zstyle ':omz:update' mode auto .*\)/\1/" ~/.zshrc
# 设置zshhistory规模
cat >> ~/.zshrc << EOF
HISTSIZE=1000000
SAVEHIST=1000000
EOF

cat >> ~/.oh-my-zsh/custom/aliases.zsh << EOF
alias lsdu='find . -maxdepth 1 -mindepth 1 -type d -exec du -sh {}+'
alias tree='tree --filesfirst --gitignore'
alias tree1='tree --filesfirst --gitignore --timefmt="%Y-%m-%d %H:%M:%S" -sh'
EOF
```

## 2.1 安装cargo

```bash
# 设置cargo/rustc等下载镜像
# USTC mirror
# export RUSTUP_DIST_SERVER="https://mirrors.ustc.edu.cn/rust-static"
# export RUSTUP_UPDATE_ROOT="https://mirrors.ustc.edu.cn/rust-static/rustup"
# THSINGHUA mirror
# export RUSTUP_DIST_SERVER=https://mirrors.tuna.tsinghua.edu.cn/rustup
# export RUSTUP_UPDATE_ROOT=https://mirrors.tuna.tsinghua.edu.cn/rustup
# ALIYUN mirror
cat >> ~/.zshenv << EOF
export RUSTUP_UPDATE_ROOT=https://mirrors.aliyun.com/rustup/rustup
export RUSTUP_DIST_SERVER=https://mirrors.aliyun.com/rustup
EOF
source ~/.zshenv
# 安装cargo/rustc等
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh

# 配置Crates镜像
cat > ~/.cargo/config.toml << EOF
[source.crates-io]
replace-with = 'aliyun'

[source.aliyun]
registry = "sparse+https://mirrors.aliyun.com/crates.io-index"

[source.tuna]
registry = "https://mirrors.tuna.tsinghua.edu.cn/git/crates.io-index"

[source.ustc]
registry = "sparse+https://mirrors.ustc.edu.cn/crates.io-index"
EOF

rustup update stable
```

## 2.2 安装uv

```bash
# export UV_INDEX_URL=https://mirrors.tuna.tsinghua.edu.cn/pypi/web/simple
cat >> ~/.zshenv << EOF
export PYTHONDONTWRITEBYTECODE=1
export UV_INDEX_URL=https://mirrors.aliyun.com/pypi/simple
EOF
source ~/.zshenv
wget -qO- https://astral.sh/uv/install.sh | sh

# 支持国内镜像自动选择的版本
# curl -LsSf https://uv.agentsmirror.com/install-cn.sh | sh

# 配置项目级uv
# 参考1: https://packaging.python.org/en/latest/
# 参考2：https://packaging.python.org/en/latest/specifications/pyproject-toml/
# 配置用户级uv
# index-url = "https://mirrors.tuna.tsinghua.edu.cn/pypi/web/simple"
cat >> ~/.config/uv/config.toml << EOF
index-url = "https://mirrors.aliyun.com/pypi/simple"
EOF


# 安装常用工具
source ~/.zshrc
uv self update
uv tool install taplo
uv tool install pre-commit
```
